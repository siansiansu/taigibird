title: "1920. Build Array from Permutation"
tags: [leetcode]
---

```cpp
class Solution {
public:
    vector<int> buildArray(vector<int>& nums)
    {
        vector<int> res(nums.size());
        for (int i = 0; i < nums.size(); i++)
        {
            res[i] = nums[nums[i]];
        }
        return res;
    }
};
```

- T: $O(n)$
- S: $O(n)$
